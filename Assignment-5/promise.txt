Bonus Question:
// Your code here
const p = Promise.resolve('World');

p.then(value => {
  // return a new Promise that resolves after 1 second
  return new Promise(resolve => {
    setTimeout(() => {
      resolve('Hello ' + value);
    }, 1000);
  });
})
.then(finalValue => {
  console.log(finalValue);  // Logs: Hello World (after 1 second)
});
